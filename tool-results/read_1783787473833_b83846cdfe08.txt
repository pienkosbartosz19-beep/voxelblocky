     4→// Helper: many blocks share the same tile for all faces.
     5→function cube(name: string, tile: number, color: string, opts: Partial<BlockDef> = {}): BlockDef {
     6→  return {
     7→    id: name as BlockType,
     8→    name,
     9→    color,
    10→    tiles: { all: tile },
    11→    solid: true,
    12→    breakable: true,
    13→    tool: 'hand',
    14→    hardness: 1.0,
    15→    ...opts,
    16→  } as BlockDef;
    17→}
    18→
    19→// Helper: grass-style block (top / side / bottom different)
    20→function grassyBlock(name: string, topTile: number, sideTile: number, bottomTile: number, color: string, sideColor: string, bottomColor: string, opts: Partial<BlockDef> = {}): BlockDef {
    21→  return {
    22→    id: name as BlockType,
    23→    name,
    24→    color,
    25→    sideColor,
    26→    bottomColor,
    27→    tiles: { top: topTile, side: sideTile, bottom: bottomTile },
    28→    solid: true,
    29→    breakable: true,
    30→    tool: 'shovel',
    31→    hardness: 0.6,
    32→    ...opts,
    33→  } as BlockDef;
    34→}
    35→
    36→// Helper: wood block (bark sides + rings on top/bottom)
    37→function woodBlock(name: string, sideTile: number, topTile: number, color: string, sideColor: string): BlockDef {
    38→  return {
    39→    id: name as BlockType,
    40→    name,
    41→    color,
    42→    sideColor,
    43→    bottomColor: sideColor,
    44→    tiles: { top: topTile, side: sideTile, bottom: topTile },
    45→    solid: true,
    46→    breakable: true,
    47→    tool: 'axe',
    48→    hardness: 1.2,
    49→    isWood: true,
    50→  } as BlockDef;
    51→}
    52→
    53→// Helper: leaves
    54→function leavesBlock(name: string, tile: number, color: string): BlockDef {
    55→  return {
    56→    id: name as BlockType,
    57→    name,
    58→    color,
    59→    tiles: { all: tile },
    60→    solid: true,
    61→    breakable: true,

    66→    drops: [{ item: name as BlockType, count: 1 }],
    67→  } as BlockDef;
    68→}

    70→// Helper: cross-shape plant
    71→function crossPlant(name: string, tile: number, color: string, hardness = 0.1): BlockDef {
    72→  return {
    73→    id: name as BlockType,
    74→    name,
    75→    color,
    76→    tiles: { all: tile },
    77→    solid: false,
    78→    breakable: true,
    79→    tool: 'hand',
    80→    hardness,
    81→    transparent: true,
    82→    variant: 'cross',
    83→    drops: [{ item: name as BlockType, count: 1 }],
    84→  } as BlockDef;
    85→}
