   307→// Helper: deterministic PRNG per tile (so each tile is stable across reloads)
   308→function tileRng(tileIndex: number) {
   309→  let a = (tileIndex * 2654435761) >>> 0;
   310→  return function () {
   311→    a |= 0;
   312→    a = (a + 0x6D2B79F5) | 0;
   313→    let t = a;
   314→    t = Math.imul(t ^ (t >>> 15), t | 1);
   315→    t ^= t + Math.imul(t ^ (t >>> 7), t | 61);
   316→    return ((t ^ (t >>> 14)) >>> 0) / 4294967296;
   317→  };
   318→}

   320→function hexToRgb(hex: string): [number, number, number] {
   321→  const h = hex.replace('#', '');
   322→  return [
   323→    parseInt(h.substring(0, 2), 16),
   324→    parseInt(h.substring(2, 4), 16),
   325→    parseInt(h.substring(4, 6), 16),
   326→  ];
   327→}

   329→function mix(a: [number, number, number], b: [number, number, number], t: number): [number, number, number] {
   330→  return [
   331→    Math.round(a[0] * (1 - t) + b[0] * t),
   332→    Math.round(a[1] * (1 - t) + b[1] * t),
   333→    Math.round(a[2] * (1 - t) + b[2] * t),
   334→  ];
   335→}
   336→
   337→function shade(c: [number, number, number], f: number): [number, number, number] {
   338→  return [
   339→    Math.max(0, Math.min(255, Math.round(c[0] * f))),
   340→    Math.max(0, Math.min(255, Math.round(c[1] * f))),
   341→    Math.max(0, Math.min(255, Math.round(c[2] * f))),
   342→  ];
   343→}
   344→
   345→type RGB = [number, number, number];
   346→
   347→interface TileCtx {
   348→  ctx: CanvasRenderingContext2D;
   349→  ox: number; // origin x in atlas pixels
   350→  oy: number; // origin y in atlas pixels
   351→  rng: () => number;
   352→}
   353→
   354→function setPx(t: TileCtx, x: number, y: number, c: RGB, a = 1) {
   355→  t.ctx.fillStyle = `rgba(${c[0]},${c[1]},${c[2]},${a})`;
   356→  t.ctx.fillRect(t.ox + x, t.oy + y, 1, 1);
   357→}
   358→
   359→// Fill base color
   360→function fillBase(t: TileCtx, c: RGB) {
   361→  t.ctx.fillStyle = `rgb(${c[0]},${c[1]},${c[2]})`;
   362→  t.ctx.fillRect(t.ox, t.oy, TILE_SIZE, TILE_SIZE);
   363→}
   364→
   365→// Add per-pixel noise (vary brightness)
   366→function addNoise(t: TileCtx, baseColor: RGB, variance: number) {
   367→  for (let y = 0; y < TILE_SIZE; y++) {
   368→    for (let x = 0; x < TILE_SIZE; x++) {
   369→      const n = (t.rng() - 0.5) * variance;
   370→      setPx(t, x, y, shade(baseColor, 1 + n));
   371→    }
   372→  }
   373→}
   374→
   375→// Sprinkle spots of a color at given density
   376→function sprinkle(t: TileCtx, color: RGB, density: number) {
   377→  for (let y = 0; y < TILE_SIZE; y++) {
   378→    for (let x = 0; x < TILE_SIZE; x++) {
   379→      if (t.rng() < density) {
   380→        setPx(t, x, y, color);
   381→      }
   382→    }
   383→  }
   384→}
   385→
   386→// Sprinkle with darker/lighter variants of base color
   387→function sprinkleShaded(t: TileCtx, base: RGB, density: number, factor: number) {
   388→  const c = shade(base, factor);
   389→  for (let y = 0; y < TILE_SIZE; y++) {
   390→    for (let x = 0; x < TILE_SIZE; x++) {
   391→      if (t.rng() < density) {
   392→        setPx(t, x, y, c);
   393→      }
   394→    }
   395→  }
   396→}
   397→
   398→// Draw a small blob (used for ore nuggets)
   399→function blob(t: TileCtx, cx: number, cy: number, r: number, color: RGB) {
   400→  for (let y = -r; y <= r; y++) {
   401→    for (let x = -r; x <= r; x++) {
   402→      if (x * x + y * y <= r * r) {
   403→        const px = cx + x;
   404→        const py = cy + y;
   405→        if (px < 0 || px >= TILE_SIZE || py < 0 || py >= TILE_SIZE) continue;
   406→        const jitter = (t.rng() - 0.5) * 0.4;
   407→        setPx(t, px, py, shade(color, 1 + jitter));
   408→      }
   409→    }
   410→  }
   411→}

   465→    const len = 3 + Math.floor(t.rng() * 4);
   466→    for (let j = 0; j < len; j++) {

   468→        setPx(t, x, y, [90, 90, 90]);
   469→      }
   470→      x += t.rng() > 0.5 ? 1 : -1;
   471→      y += t.rng() > 0.5 ? 1 : 0;
   472→    }
   473→  }

   476→function pCobblestone(t: TileCtx) {
   477→  fillBase(t, [110, 110, 110]);
   478→  // Draw rounded stones with mortar lines
   479→  const stones: { x: number; y: number; r: number }[] = [];
   480→  for (let i = 0; i < 5; i++) {
   481→    stones.push({
   482→      x: 2 + Math.floor(t.rng() * 12),
   483→      y: 2 + Math.floor(t.rng() * 12),
   484→      r: 2 + Math.floor(t.rng() * 2),
   485→    });
   486→  }
   487→  // Dark mortar background
   488→  for (let y = 0; y < TILE_SIZE; y++) {
   489→    for (let x = 0; x < TILE_SIZE; x++) {
   490→      let inStone = false;
   491→      for (const s of stones) {
   492→        if ((x - s.x) * (x - s.x) + (y - s.y) * (y - s.y) <= s.r * s.r) {
   493→          inStone = true;
   494→          const n = (t.rng() - 0.5) * 0.2;
   495→          setPx(t, x, y, shade([125, 125, 125], 1 + n));
   496→          break;
   497→        }
   498→      }
   499→      if (!inStone) {
   500→        const n = (t.rng() - 0.5) * 0.1;
   501→        setPx(t, x, y, shade([75, 75, 75], 1 + n));
   502→      }
   503→    }
   504→  }
   505→  // Highlights on top-left of each stone
   506→  for (const s of stones) {
   507→    setPx(t, s.x - 1, s.y - 1, [170, 170, 170]);
   508→    setPx(t, s.x, s.y - 1, [160, 160, 160]);
   509→  }
   510→}

   543→function pGravel(t: TileCtx) {

  1706→// Build the atlas into a single canvas. Returns the canvas.
  1707→export function buildAtlasCanvas(): HTMLCanvasElement {
  1708→  const canvas = document.createElement('canvas');
  1709→  canvas.width = ATLAS_W;
  1710→  canvas.height = ATLAS_H;
  1711→  const ctx = canvas.getContext('2d')!;
  1712→  ctx.imageSmoothingEnabled = false;
  1713→
  1714→  for (let tileIdx = 0; tileIdx < 256; tileIdx++) {
  1715→    const col = tileIdx % ATLAS_COLS;
  1716→    const row = Math.floor(tileIdx / ATLAS_COLS);
  1717→    const ox = col * TILE_SIZE;
  1718→    const oy = row * TILE_SIZE;
  1719→    const t: TileCtx = { ctx, ox, oy, rng: tileRng(tileIdx) };
  1720→
  1721→    const painter = TILE_PAINTERS[tileIdx];
  1722→    if (painter) {
  1723→      painter(t);
  1724→    } else {
  1725→      // No painter - render as missing
  1726→      pMissing(t);
  1727→    }
  1728→  }
  1729→
  1730→  return canvas;
  1731→}

  1733→// Compute UV rectangle for a tile index, padded to avoid bleeding.
  1734→// Returns [u0, v0, u1, v1] in atlas space [0..1].
  1735→export function tileUV(tileIdx: number): [number, number, number, number] {
  1736→  const col = tileIdx % ATLAS_COLS;
  1737→  const row = Math.floor(tileIdx / ATLAS_COLS);
  1738→  const pad = 0.5 / ATLAS_W; // half-texel padding to prevent bleed
  1739→  const u0 = (col * TILE_SIZE) / ATLAS_W + pad;
  1740→  const v1 = 1 - (row * TILE_SIZE) / ATLAS_H - pad;
  1741→  const u1 = ((col + 1) * TILE_SIZE) / ATLAS_W - pad;
  1742→  const v0 = 1 - ((row + 1) * TILE_SIZE) / ATLAS_H + pad;
  1743→  return [u0, v0, u1, v1];
  1744→}

  1746→// Build a THREE.CanvasTexture from the atlas. Caches the result.
  1747→let cachedTexture: THREE.CanvasTexture | null = null;
  1748→
  1749→export function getAtlasTexture(): THREE.CanvasTexture {
  1750→  if (cachedTexture) return cachedTexture;
  1751→  const canvas = buildAtlasCanvas();
  1752→  const tex = new THREE.CanvasTexture(canvas);
  1753→  tex.magFilter = THREE.NearestFilter;
  1754→  tex.minFilter = THREE.NearestFilter;
  1755→  tex.generateMipmaps = false;
  1756→  tex.colorSpace = THREE.SRGBColorSpace;
  1757→  tex.needsUpdate = true;
  1758→  cachedTexture = tex;
  1759→  return tex;
  1760→}
