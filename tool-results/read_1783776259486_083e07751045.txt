   181→    function addToInventory(block: BlockType, count = 1): number {
   182→      if (block === 'air') return 0;
   183→      // First, try hotbar slots
   184→      let remaining = count;
   185→      // Try to stack into existing inventory slots
   186→      for (let i = 0; i < inventory.length && remaining > 0; i++) {
   187→        if (inventory[i].block === block) {
   188→          inventory[i].count += remaining;
   189→          remaining = 0;
   190→        }
   191→      }
   192→      // Find empty inventory slot
   193→      for (let i = 0; i < inventory.length && remaining > 0; i++) {
   194→        if (inventory[i].count === 0 || inventory[i].block === 'air') {
   195→          inventory[i].block = block;
   196→          inventory[i].count = remaining;
   197→          remaining = 0;
   198→        }
   199→      }
   200→      updateQuestProgress(block, count);
   201→      return count - remaining;
   202→    }

   204→    function removeFromInventory(block: BlockType, count = 1): boolean {
   205→      // Count available
   206→      let available = 0;
   207→      for (const item of inventory) {
   208→        if (item.block === block) available += item.count;
   209→      }
   210→      if (available < count) return false;
   211→
   212→      let remaining = count;
   213→      for (let i = 0; i < inventory.length && remaining > 0; i++) {
   214→        if (inventory[i].block === block) {
   215→          const take = Math.min(inventory[i].count, remaining);
   216→          inventory[i].count -= take;
   217→          remaining -= take;
   218→          if (inventory[i].count === 0) inventory[i].block = 'air';
   219→        }
   220→      }
   221→      return true;
   222→    }

   235→    function updateQuestProgress(block: BlockType, count: number) {
   236→      for (const q of questState) {

   290→        player.velocity.y = 0;

   301→    function onMouseDown(e: MouseEvent) {
   302→      // If pointer not locked, try to lock it on first click

   304→        canvas.requestPointerLock();

   306→      }
   307→      if (e.button === 0) {
   308→        leftMouseDown = true;
   309→        stateRef.current.breaking = true;
   310→        syncUI();
   311→      } else if (e.button === 2) {
   312→        rightMouseDown = true;
   313→        // Place block

   316→      }

   331→    function onMouseMove(e: MouseEvent) {
   332→      // Use movementX/Y if available (works with or without pointer lock)
   333→      const mx = e.movementX || 0;
   334→      const my = e.movementY || 0;
   335→      if (mx === 0 && my === 0) return;
   336→      // Only rotate camera if pointer locked OR mouse is over canvas
   337→      if (!mouseLocked && document.pointerLockElement !== canvas) return;
   338→      const sensitivity = 0.0025;
   339→      player.yaw -= mx * sensitivity;
   340→      player.pitch -= my * sensitivity;
   341→      player.pitch = Math.max(-Math.PI / 2 + 0.01, Math.min(Math.PI / 2 - 0.01, player.pitch));
   342→    }

   399→    function tryBreakBlock() {
   400→      const hit = getLookingAt();
   401→      if (!hit.hit) {
   402→        breakingBlock = null;
   403→        breakProgress = 0;
   404→        return;
   405→      }
   406→
   407→      const def = BLOCKS[hit.blockType];
   408→      if (!def.breakable) {
   409→        setMessage('Nie można zniszczyć tego bloku!', 1500);
   410→        breakingBlock = null;
   411→        return;
   412→      }
   413→
   414→      // If new block, reset progress
   415→      if (!breakingBlock || breakingBlock.x !== hit.block.x || breakingBlock.y !== hit.block.y || breakingBlock.z !== hit.block.z) {
   416→        breakingBlock = { ...hit.block };
   417→        breakProgress = 0;
   418→      }
   419→
   420→      breakProgress += 1 / 60; // approx
   421→      const breakTime = def.hardness;
   422→      if (breakProgress >= breakTime) {
   423→        // Break!
   424→        world.setBlock(hit.block.x, hit.block.y, hit.block.z, 'air');
   425→        // Drops
   426→        if (def.drops) {
   427→          for (const drop of def.drops) {
   428→            addToInventory(drop.item, drop.count);
   429→          }
   430→        }
   431→        // Mark surrounding chunks for rebuild
   432→        markChunksDirty(hit.block.x, hit.block.y, hit.block.z);
   433→        breakingBlock = null;
   434→        breakProgress = 0;
   435→      }
   436→    }

   438→    function tryPlaceBlock() {
   439→      const hit = getLookingAt();
   440→      if (!hit.hit) return;
   441→
   442→      // Selected block from hotbar
   443→      const blockToPlace = hotbar[selectedSlot];
   444→      if (!blockToPlace) return;
   445→      const def = BLOCKS[blockToPlace];
   446→      if (def.variant === 'cross' || blockToPlace === 'torch') {
   447→        // place at adjacent
   448→      }
   449→
   450→      // Count in inventory
   451→      if (getInventoryCount(blockToPlace) <= 0) {
   452→        setMessage(`Brak ${def.name} w ekwipunku`, 1200);
   453→        return;
   454→      }
   455→
   456→      // Place at adjacent position
   457→      const px = hit.block.x + hit.normal.x;
   458→      const py = hit.block.y + hit.normal.y;
   459→      const pz = hit.block.z + hit.normal.z;
   460→
   461→      // Don't place inside player
   462→      const playerMinX = player.position.x - 0.3;
   463→      const playerMaxX = player.position.x + 0.3;
   464→      const playerMinY = player.position.y;
   465→      const playerMaxY = player.position.y + 1.8;
   466→      const playerMinZ = player.position.z - 0.3;
   467→      const playerMaxZ = player.position.z + 0.3;
   468→      if (
   469→        px + 1 > playerMinX && px < playerMaxX &&
   470→        py + 1 > playerMinY && py < playerMaxY &&
   471→        pz + 1 > playerMinZ && pz < playerMaxZ &&
   472→        def.solid
   473→      ) {
   474→        return;
   475→      }
   476→
   477→      // Check existing block
   478→      const existing = world.getBlock(px, py, pz);
   479→      if (existing !== 'air' && existing !== 'water') return;
   480→
   481→      world.setBlock(px, py, pz, blockToPlace);
   482→      removeFromInventory(blockToPlace, 1);
   483→      markChunksDirty(px, py, pz);
   484→    }

   486→    function markChunksDirty(x: number, y: number, z: number) {
   487→      // Determine affected chunks
   488→      const cx = Math.floor(x / 16);
   489→      const cz = Math.floor(z / 16);
   490→      const lx = ((x % 16) + 16) % 16;
   491→      const lz = ((z % 16) + 16) % 16;
   492→      chunkManager.rebuildChunk(cx, cz);
   493→      if (lx === 0) chunkManager.rebuildChunk(cx - 1, cz);
   494→      if (lx === 15) chunkManager.rebuildChunk(cx + 1, cz);
   495→      if (lz === 0) chunkManager.rebuildChunk(cx, cz - 1);
   496→      if (lz === 15) chunkManager.rebuildChunk(cx, cz + 1);
   497→    }

   500→    function checkCollision(x: number, y: number, z: number): boolean {
   501→      const minX = Math.floor(x - 0.3);
   502→      const maxX = Math.floor(x + 0.3);
   503→      const minY = Math.floor(y);
   504→      const maxY = Math.floor(y + 1.79);
   505→      const minZ = Math.floor(z - 0.3);
   506→      const maxZ = Math.floor(z + 0.3);
   507→      for (let bx = minX; bx <= maxX; bx++) {
   508→        for (let by = minY; by <= maxY; by++) {
   509→          for (let bz = minZ; bz <= maxZ; bz++) {
   510→            const bt = world.getBlock(bx, by, bz);
   511→            if (BLOCKS[bt].solid) return true;
   512→          }
   513→        }
   514→      }
   515→      return false;
   516→    }

   524→    function updatePlayer(dt: number) {
   525→      if (stateRef.current.dead) return;
   526→
   527→      // Stamina regen
   528→      if (!leftMouseDown && stateRef.current.stamina < stateRef.current.maxStamina) {
   529→        stateRef.current.stamina = Math.min(stateRef.current.maxStamina, stateRef.current.stamina + dt * 12);
   530→      }
   531→
   532→      // Hunger drain
   533→      const moving = keys['KeyW'] || keys['KeyA'] || keys['KeyS'] || keys['KeyD'];
   534→      const sprinting = moving && keys['ShiftLeft'] && stateRef.current.stamina > 0;
   535→      const hungerRate = sprinting ? 0.15 : moving ? 0.05 : 0.02;
   536→      stateRef.current.hunger = Math.max(0, stateRef.current.hunger - dt * hungerRate);
   537→
   538→      // Health regen if fed
   539→      if (stateRef.current.hunger > 15 && stateRef.current.health < stateRef.current.maxHealth) {
   540→        stateRef.current.health = Math.min(stateRef.current.maxHealth, stateRef.current.health + dt * 0.5);
   541→      } else if (stateRef.current.hunger === 0) {
   542→        // Starving
   543→        stateRef.current.health = Math.max(0, stateRef.current.health - dt * 0.5);
   544→        if (stateRef.current.health <= 0) {
   545→          stateRef.current.dead = true;
   546→          setMessage('Umarłeś z głodu! Kliknij aby zrespawnować.', 99999);
   547→          syncUI();
   548→          return;
   549→        }
   550→      }
   551→
   552→      // Movement input
   553→      const speed = sprinting ? WORLD_CONFIG.SPRINT_SPEED : WORLD_CONFIG.WALK_SPEED;
   554→      const yaw = player.yaw;
   555→      const forward = new THREE.Vector3(-Math.sin(yaw), 0, -Math.cos(yaw));
   556→      const right = new THREE.Vector3(Math.cos(yaw), 0, -Math.sin(yaw));
   557→      const moveDir = new THREE.Vector3();
   558→      if (keys['KeyW']) moveDir.add(forward);
   559→      if (keys['KeyS']) moveDir.sub(forward);
   560→      if (keys['KeyD']) moveDir.add(right);
   561→      if (keys['KeyA']) moveDir.sub(right);
   562→      if (moveDir.lengthSq() > 0) {
   563→        moveDir.normalize().multiplyScalar(speed);
   564→      }
   565→
   566→      if (sprinting) {
   567→        stateRef.current.stamina = Math.max(0, stateRef.current.stamina - dt * 15);
   568→      }
   569→
   570→      if (player.flying) {
   571→        player.velocity.x = moveDir.x;
   572→        player.velocity.z = moveDir.z;
   573→        player.velocity.y = 0;
   574→        if (keys['Space']) player.velocity.y = WORLD_CONFIG.FLY_SPEED * 0.7;
   575→        if (keys['ShiftLeft']) player.velocity.y = -WORLD_CONFIG.FLY_SPEED * 0.7;
   576→      } else {
   577→        player.velocity.x = moveDir.x;
   578→        player.velocity.z = moveDir.z;
   579→
   580→        // Gravity
   581→        player.velocity.y += WORLD_CONFIG.GRAVITY * dt;
   582→        if (player.velocity.y < -50) player.velocity.y = -50;
   583→
   584→        // Water physics
   585→        if (isInWater(player.position.x, player.position.y, player.position.z)) {
   586→          player.velocity.y = Math.max(player.velocity.y, -3);
   587→          player.velocity.y *= 0.85;
   588→          if (keys['Space']) player.velocity.y = 4;
   589→        }
   590→
   591→        // Jump
   592→        if (keys['Space'] && player.onGround && stateRef.current.stamina > 5) {
   593→          player.velocity.y = WORLD_CONFIG.JUMP_VELOCITY;
   594→          player.onGround = false;
   595→          stateRef.current.stamina -= 3;
   596→        }
   597→      }
   598→
   599→      // Apply movement with collision (per-axis)
   600→      const dx = player.velocity.x * dt;
   601→      const dy = player.velocity.y * dt;
   602→      const dz = player.velocity.z * dt;
   603→
   604→      // X axis
   605→      if (dx !== 0 && !checkCollision(player.position.x + dx, player.position.y, player.position.z)) {
   606→        player.position.x += dx;
   607→      } else {
   608→        player.velocity.x = 0;
   609→      }
   610→      // Z axis
   611→      if (dz !== 0 && !checkCollision(player.position.x, player.position.y, player.position.z + dz)) {
   612→        player.position.z += dz;
   613→      } else {
   614→        player.velocity.z = 0;
   615→      }
   616→      // Y axis
   617→      if (dy !== 0 && !checkCollision(player.position.x, player.position.y + dy, player.position.z)) {
   618→        player.position.y += dy;
   619→        player.onGround = false;
   620→      } else {
   621→        if (dy < 0) player.onGround = true;
   622→        player.velocity.y = 0;
   623→      }
   624→
   625→      // Fall damage / void
   626→      if (player.position.y < -10) {
   627→        stateRef.current.health = 0;
   628→        stateRef.current.dead = true;
   629→        setMessage('Spadłeś w otchłań! Kliknij aby zrespawnować.', 99999);
   630→        syncUI();
   631→      }
   632→
   633→      // Update camera position
   634→      camera.position.set(player.position.x, player.position.y + 1.6, player.position.z);
   635→      const euler = new THREE.Euler(player.pitch, player.yaw, 0, 'YXZ');
   636→      camera.quaternion.setFromEuler(euler);
   637→
   638→      // Breaking
   639→      if (leftMouseDown) {
   640→        if (placeCooldown > 0) placeCooldown -= dt;
   641→        tryBreakBlock();
   642→      }
   643→
   644→      stateRef.current.position = { x: player.position.x, y: player.position.y, z: player.position.z };
   645→    }
   646→
   647→    // === Day/night cycle ===
   648→    const dayLength = 240; // 4 minutes per day
   649→    let time = 0.35; // start morning
   650→
   651→    function updateDayNight(dt: number) {
   652→      time = (time + dt / dayLength) % 1;
   653→      stateRef.current.time = time;
   654→      stateRef.current.isDay = time > 0.22 && time < 0.78;
   655→
   656→      // Sun position
   657→      const sunAngle = (time - 0.25) * Math.PI * 2;
   658→      const sunHeight = Math.sin(sunAngle);
   659→      const sunX = Math.cos(sunAngle) * 80;
   660→      sun.position.set(sunX, sunHeight * 80 + 5, 30);
   661→
   662→      // Light intensity
   663→      const dayFactor = Math.max(0, Math.sin(sunAngle));
   664→      sun.intensity = 0.2 + dayFactor * 1.0;
   665→      ambient.intensity = 0.25 + dayFactor * 0.35;
   666→      hemi.intensity = 0.2 + dayFactor * 0.3;
   667→
   668→      // Moon
   669→      moon.position.set(-sunX, -sunHeight * 80 + 5, -30);
   670→      moon.intensity = (1 - dayFactor) * 0.3;
   671→
   672→      // Sky color
   673→      const dayColor = new THREE.Color('#9fc8ec');
   674→      const nightColor = new THREE.Color('#0a1535');
   675→      const sunsetColor = new THREE.Color('#f4a261');
   676→      const skyColor = new THREE.Color();
   677→      if (dayFactor > 0.3) {
   678→        skyColor.copy(dayColor);
   679→      } else if (dayFactor > 0) {
   680→        // sunrise/sunset blend
   681→        const t = dayFactor / 0.3;
   682→        skyColor.copy(sunsetColor).lerp(dayColor, t);
   683→      } else {
   684→        skyColor.copy(nightColor);
   685→      }
   686→      scene.background = skyColor;
   687→      if (scene.fog) {
   688→        (scene.fog as THREE.Fog).color.copy(skyColor);
   689→      }
   690→    }
   691→
   692→    // === Block highlight (wireframe box at looked-at block) ===
   693→    const highlightGeo = new THREE.BoxGeometry(1.001, 1.001, 1.001);
   694→    const highlightEdges = new THREE.EdgesGeometry(highlightGeo);
   695→    const highlightMat = new THREE.LineBasicMaterial({ color: 0xffffff, transparent: true, opacity: 0.6 });
   696→    const highlight = new THREE.LineSegments(highlightEdges, highlightMat);
   697→    highlight.visible = false;
   698→    scene.add(highlight);
   699→
   700→    // Break overlay
   701→    const breakOverlayGeo = new THREE.BoxGeometry(1.02, 1.02, 1.02);
   702→    const breakOverlayMat = new THREE.MeshBasicMaterial({
   703→      color: 0x000000,
   704→      transparent: true,
   705→      opacity: 0,
   706→      depthWrite: false,
   707→    });
   708→    const breakOverlay = new THREE.Mesh(breakOverlayGeo, breakOverlayMat);
   709→    breakOverlay.visible = false;
   710→    scene.add(breakOverlay);
   711→
   712→    // === Main loop ===
   713→    let lastTime = performance.now();
   714→    let frameCount = 0;
   715→    let fpsTime = 0;
   716→    let fps = 0;
   717→    let uiSyncTimer = 0;
   718→    let chunkUpdateAccum = 0;
   719→    let running = true;
   720→
   721→    function frame() {
   722→      if (!running) return;
   723→      requestAnimationFrame(frame);
   724→
   725→      const now = performance.now();
   726→      const dt = Math.min(0.05, (now - lastTime) / 1000);
   727→      lastTime = now;
   728→      frameCount++;
   729→      fpsTime += dt;
   730→      if (fpsTime > 0.5) {
   731→        fps = Math.round(frameCount / fpsTime);
   732→        frameCount = 0;
   733→        fpsTime = 0;
   734→        stateRef.current.fps = fps;
   735→      }
   736→
   737→      try {
   738→        if (!stateRef.current.paused && !stateRef.current.dead) {
   739→          updatePlayer(dt);
   740→          updateDayNight(dt);
   741→        }
   742→
   743→        // Update chunks - only when not paused (player has pointer lock)
   744→        // and throttle to spread cost across frames
   745→        if (!stateRef.current.paused) {
   746→          chunkUpdateAccum += dt;
   747→          if (chunkUpdateAccum > 0.25) {
   748→            chunkUpdateAccum = 0;
   749→            chunkManager.update(player.position.x, player.position.z, 1);
   750→          }
   751→        }
   752→        stateRef.current.chunkCount = chunkManager.chunks.size;
   753→        stateRef.current.triangleCount = Math.round(chunkManager.triangleCount);
   754→
   755→      // Update highlight
   756→      const hit = getLookingAt();
   757→      if (hit.hit) {
   758→        highlight.visible = true;
   759→        highlight.position.set(hit.block.x + 0.5, hit.block.y + 0.5, hit.block.z + 0.5);
   760→        stateRef.current.lookingAt = hit.blockType;
   761→      } else {
   762→        highlight.visible = false;
   763→        stateRef.current.lookingAt = null;
   764→      }
   765→
   766→      // Break overlay
   767→      if (leftMouseDown && breakingBlock) {
   768→        breakOverlay.visible = true;
   769→        breakOverlay.position.set(breakingBlock.x + 0.5, breakingBlock.y + 0.5, breakingBlock.z + 0.5);
   770→        const def = BLOCKS[world.getBlock(breakingBlock.x, breakingBlock.y, breakingBlock.z)];
   771→        const progress = breakProgress / Math.max(0.1, def.hardness);
   772→        breakOverlayMat.opacity = Math.min(0.5, progress * 0.5);
   773→      } else {
   774→        breakOverlay.visible = false;
   775→      }
   776→
   777→      // Determine biome at player position
   778→      const wx = Math.floor(player.position.x);
   779→      const wz = Math.floor(player.position.z);
   780→      const groundBlock = world.getBlock(wx, Math.floor(player.position.y - 0.5), wz);
   781→      let biome = 'plains';
   782→      if (groundBlock === 'sand') biome = 'pustynia';
   783→      else if (groundBlock === 'snow') biome = 'tundra';
   784→      else if (groundBlock === 'stone') biome = 'góry';
   785→      else if (groundBlock === 'water') biome = 'woda';
   786→      else if (groundBlock === 'grass') biome = 'łąka';
   787→      stateRef.current.biome = biome;
   788→
   789→      // Sync UI (throttled)
   790→      uiSyncTimer += dt;
   791→      if (uiSyncTimer > 0.1) {
   792→        uiSyncTimer = 0;
   793→        // Push hotbar & inventory snapshot to UI
   794→        stateRef.current.hotbar = [...hotbar];
   795→        stateRef.current.inventory = inventory.filter(i => i.count > 0 && i.block !== 'air');
   796→        syncUI();
   797→      }
   798→
   799→      renderer.render(scene, camera);
   800→      } catch (err) {
   801→        console.error('Game frame error:', err);
   802→      }
   803→    }
   804→
   805→    frame();
   806→
   807→    // === Public control API ===
   808→    gameRef.current = {
   809→      start: () => {
   810→        canvas.requestPointerLock();
   811→      },
   812→      respawn: () => {
   813→        const s = world.findSpawn();
   814→        player.position = { x: s.x, y: s.y, z: s.z };
   815→        player.velocity = { x: 0, y: 0, z: 0 };
   816→        stateRef.current.health = 20;
   817→        stateRef.current.hunger = 20;
   818→        stateRef.current.stamina = 100;
   819→        stateRef.current.dead = false;
   820→        stateRef.current.message = null;
   821→        syncUI();
   822→      },
   823→      setHotbarSlot: (slot: number, block: BlockType | null) => {
   824→        if (slot >= 0 && slot < HOTBAR_SLOTS) {
   825→          hotbar[slot] = block;
   826→        }
   827→      },
   828→      getInventory: () => inventory,
   829→      getHotbar: () => hotbar,
   830→      moveItemToHotbar: (invIndex: number, hotbarSlot: number) => {
   831→        if (invIndex < 0 || invIndex >= inventory.length) return;
   832→        if (hotbarSlot < 0 || hotbarSlot >= HOTBAR_SLOTS) return;
   833→        const invItem = inventory[invIndex];
   834→        if (invItem.count === 0 || invItem.block === 'air') return;
   835→        const current = hotbar[hotbarSlot];
   836→        hotbar[hotbarSlot] = invItem.block;
   837→        if (current) {
   838→          invItem.block = current;
   839→        } else {
   840→          invItem.block = 'air';
   841→          invItem.count = 0;
   842→        }
   843→      },
   844→      addToInventoryFromUI: (block: BlockType, count: number) => addToInventory(block, count),
   845→      craftItem: (recipeId: string, recipes: any[]) => {
   846→        const recipe = recipes.find(r => r.id === recipeId);
   847→        if (!recipe) return false;
   848→        // Check ingredients
   849→        for (const ing of recipe.ingredients) {
   850→          if (getInventoryCount(ing.block) < ing.count) {
   851→            setMessage(`Brak składników: ${BLOCKS[ing.block].name}`, 1800);
   852→            return false;
   853→          }
   854→        }
   855→        // Consume
   856→        for (const ing of recipe.ingredients) {
   857→          removeFromInventory(ing.block, ing.count);
   858→        }
   859→        // Add output (if it's a tool, simulate by adding the block as a marker)
   860→        addToInventory(recipe.output.block, recipe.output.count);
   861→        setMessage(`Wytworzono: ${recipe.name}`, 1500);
   862→        syncUI();
   863→        return true;
   864→      },
   865→      giveItem: (block: BlockType, count: number) => {
   866→        addToInventory(block, count);
   867→        setMessage(`Dodano ${count}× ${BLOCKS[block].name}`, 1200);
   868→        syncUI();
   869→      },
   870→      getState: () => stateRef.current,
   871→    };
   872→
   873→    // Cleanup
   874→    return () => {
   875→      running = false;
   876→      document.removeEventListener('keydown', onKeyDown);
   877→      document.removeEventListener('keyup', onKeyUp);
   878→      document.removeEventListener('mousedown', onMouseDown);
   879→      document.removeEventListener('mouseup', onMouseUp);
   880→      document.removeEventListener('mousemove', onMouseMove);
   881→      document.removeEventListener('wheel', onWheel);
   882→      document.removeEventListener('pointerlockchange', onPointerLockChange);
   883→      window.removeEventListener('resize', onResize);
   884→      renderer.dispose();
   885→    };
   886→  }, [canvasRef, syncUI, setMessage, enabled]);
   887→
   888→  const startGame = useCallback(() => {
   889→    gameRef.current?.start();
   890→  }, []);
   891→
   892→  const respawn = useCallback(() => {
   893→    gameRef.current?.respawn();
   894→  }, []);
   895→
   896→  const setHotbarSlot = useCallback((slot: number, block: BlockType | null) => {
   897→    gameRef.current?.setHotbarSlot(slot, block);
   898→  }, []);
   899→
   900→  const moveItemToHotbar = useCallback((invIndex: number, hotbarSlot: number) => {
   901→    gameRef.current?.moveItemToHotbar(invIndex, hotbarSlot);
   902→  }, []);
   903→
   904→  const craftItem = useCallback((recipeId: string, recipes: any[]) => {
   905→    return gameRef.current?.craftItem(recipeId, recipes);
   906→  }, []);
   907→
   908→  const giveItem = useCallback((block: BlockType, count: number) => {
   909→    gameRef.current?.giveItem(block, count);
   910→  }, []);
   911→
   912→  return {
   913→    uiState,
   914→    startGame,
   915→    respawn,
   916→    setHotbarSlot,
   917→    moveItemToHotbar,
   918→    craftItem,
   919→    giveItem,
   920→  };
   921→}
