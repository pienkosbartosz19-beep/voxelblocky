   181→   181→    function addToInventory(block: BlockType, count = 1): number {
   182→   182→      if (block === 'air') return 0;

   184→   184→      let remaining = count;

   186→   186→      for (let i = 0; i < inventory.length && remaining > 0; i++) {
   187→   187→        if (inventory[i].block === block) {
   188→   188→          inventory[i].count += remaining;
   189→   189→          remaining = 0;
   190→   190→        }
   191→   191→      }
   192→   192→      // Find empty inventory slot
   193→   193→      for (let i = 0; i < inventory.length && remaining > 0; i++) {
   194→   194→        if (inventory[i].count === 0 || inventory[i].block === 'air') {
   195→   195→          inventory[i].block = block;
   196→   196→          inventory[i].count = remaining;
   197→   197→          remaining = 0;
   198→   198→        }
   199→   199→      }
   200→   200→      updateQuestProgress(block, count);
   201→   201→      return count - remaining;
   202→   202→    }

   204→   204→    function removeFromInventory(block: BlockType, count = 1): boolean {
   205→   205→      // Count available
   206→   206→      let available = 0;
   207→   207→      for (const item of inventory) {
   208→   208→        if (item.block === block) available += item.count;
   209→   209→      }
   210→   210→      if (available < count) return false;
   211→   211→
   212→   212→      let remaining = count;
   213→   213→      for (let i = 0; i < inventory.length && remaining > 0; i++) {
   214→   214→        if (inventory[i].block === block) {
   215→   215→          const take = Math.min(inventory[i].count, remaining);
   216→   216→          inventory[i].count -= take;
   217→   217→          remaining -= take;
   218→   218→          if (inventory[i].count === 0) inventory[i].block = 'air';
   219→   219→        }
   220→   220→      }
   221→   221→      return true;
   222→   222→    }

   399→   399→    function tryBreakBlock() {
   400→   400→      const hit = getLookingAt();
   401→   401→      if (!hit.hit) {
   402→   402→        breakingBlock = null;
   403→   403→        breakProgress = 0;
   404→   404→        return;
   405→   405→      }
   406→   406→
   407→   407→      const def = BLOCKS[hit.blockType];
   408→   408→      if (!def.breakable) {
   409→   409→        setMessage('Nie można zniszczyć tego bloku!', 1500);
   410→   410→        breakingBlock = null;
   411→   411→        return;
   412→   412→      }
   413→   413→
   414→   414→      // If new block, reset progress
   415→   415→      if (!breakingBlock || breakingBlock.x !== hit.block.x || breakingBlock.y !== hit.block.y || breakingBlock.z !== hit.block.z) {
   416→   416→        breakingBlock = { ...hit.block };
   417→   417→        breakProgress = 0;
   418→   418→      }
   419→   419→
   420→   420→      breakProgress += 1 / 60; // approx
   421→   421→      const breakTime = def.hardness;
   422→   422→      if (breakProgress >= breakTime) {
   423→   423→        // Break!
   424→   424→        world.setBlock(hit.block.x, hit.block.y, hit.block.z, 'air');
   425→   425→        // Drops
   426→   426→        if (def.drops) {
   427→   427→          for (const drop of def.drops) {
   428→   428→            addToInventory(drop.item, drop.count);
   429→   429→          }
   430→   430→        }
   431→   431→        // Mark surrounding chunks for rebuild
   432→   432→        markChunksDirty(hit.block.x, hit.block.y, hit.block.z);
   433→   433→        breakingBlock = null;
   434→   434→        breakProgress = 0;
   435→   435→      }
   436→   436→    }

   438→   438→    function tryPlaceBlock() {
   439→   439→      const hit = getLookingAt();
   440→   440→      if (!hit.hit) return;
   441→   441→
   442→   442→      // Selected block from hotbar
   443→   443→      const blockToPlace = hotbar[selectedSlot];
   444→   444→      if (!blockToPlace) return;
   445→   445→      const def = BLOCKS[blockToPlace];
   446→   446→      if (def.variant === 'cross' || blockToPlace === 'torch') {
   447→   447→        // place at adjacent
   448→   448→      }
   449→   449→
   450→   450→      // Count in inventory
   451→   451→      if (getInventoryCount(blockToPlace) <= 0) {
   452→   452→        setMessage(`Brak ${def.name} w ekwipunku`, 1200);
   453→   453→        return;
   454→   454→      }
   455→   455→
   456→   456→      // Place at adjacent position
   457→   457→      const px = hit.block.x + hit.normal.x;
   458→   458→      const py = hit.block.y + hit.normal.y;
   459→   459→      const pz = hit.block.z + hit.normal.z;
   460→   460→
   461→   461→      // Don't place inside player
   462→   462→      const playerMinX = player.position.x - 0.3;
   463→   463→      const playerMaxX = player.position.x + 0.3;
   464→   464→      const playerMinY = player.position.y;
   465→   465→      const playerMaxY = player.position.y + 1.8;
   466→   466→      const playerMinZ = player.position.z - 0.3;
   467→   467→      const playerMaxZ = player.position.z + 0.3;
   468→   468→      if (
   469→   469→        px + 1 > playerMinX && px < playerMaxX &&
   470→   470→        py + 1 > playerMinY && py < playerMaxY &&
   471→   471→        pz + 1 > playerMinZ && pz < playerMaxZ &&
   472→   472→        def.solid
   473→   473→      ) {
   474→   474→        return;

   500→   500→    function checkCollision(x: number, y: number, z: number): boolean {
   501→   501→      const minX = Math.floor(x - 0.3);
   502→   502→      const maxX = Math.floor(x + 0.3);
   503→   503→      const minY = Math.floor(y);
   504→   504→      const maxY = Math.floor(y + 1.79);
   505→   505→      const minZ = Math.floor(z - 0.3);
   506→   506→      const maxZ = Math.floor(z + 0.3);
   507→   507→      for (let bx = minX; bx <= maxX; bx++) {
   508→   508→        for (let by = minY; by <= maxY; by++) {
   509→   509→          for (let bz = minZ; bz <= maxZ; bz++) {
   510→   510→            const bt = world.getBlock(bx, by, bz);
   511→   511→            if (BLOCKS[bt].solid) return true;
   512→   512→          }
   513→   513→        }
   514→   514→      }
   515→   515→      return false;
   516→   516→    }

   524→   524→    function updatePlayer(dt: number) {
   525→   525→      if (stateRef.current.dead) return;
   526→   526→
   527→   527→      // Stamina regen
   528→   528→      if (!leftMouseDown && stateRef.current.stamina < stateRef.current.maxStamina) {
   529→   529→        stateRef.current.stamina = Math.min(stateRef.current.maxStamina, stateRef.current.stamina + dt * 12);
   530→   530→      }
   531→   531→
   532→   532→      // Hunger drain
   533→   533→      const moving = keys['KeyW'] || keys['KeyA'] || keys['KeyS'] || keys['KeyD'];
   534→   534→      const sprinting = moving && keys['ShiftLeft'] && stateRef.current.stamina > 0;
   535→   535→      const hungerRate = sprinting ? 0.15 : moving ? 0.05 : 0.02;
   536→   536→      stateRef.current.hunger = Math.max(0, stateRef.current.hunger - dt * hungerRate);
   537→   537→
   538→   538→      // Health regen if fed
   539→   539→      if (stateRef.current.hunger > 15 && stateRef.current.health < stateRef.current.maxHealth) {
   540→   540→        stateRef.current.health = Math.min(stateRef.current.maxHealth, stateRef.current.health + dt * 0.5);
   541→   541→      } else if (stateRef.current.hunger === 0) {
   542→   542→        // Starving
   543→   543→        stateRef.current.health = Math.max(0, stateRef.current.health - dt * 0.5);
   544→   544→        if (stateRef.current.health <= 0) {
   545→   545→          stateRef.current.dead = true;
   546→   546→          setMessage('Umarłeś z głodu! Kliknij aby zrespawnować.', 99999);
   547→   547→          syncUI();
   548→   548→          return;
   549→   549→        }
   550→   550→      }
   551→   551→
   552→   552→      // Movement input
   553→   553→      const speed = sprinting ? WORLD_CONFIG.SPRINT_SPEED : WORLD_CONFIG.WALK_SPEED;
   554→   554→      const yaw = player.yaw;
   555→   555→      const forward = new THREE.Vector3(-Math.sin(yaw), 0, -Math.cos(yaw));
   556→   556→      const right = new THREE.Vector3(Math.cos(yaw), 0, -Math.sin(yaw));
   557→   557→      const moveDir = new THREE.Vector3();
   558→   558→      if (keys['KeyW']) moveDir.add(forward);
   559→   559→      if (keys['KeyS']) moveDir.sub(forward);
   560→   560→      if (keys['KeyD']) moveDir.add(right);
   561→   561→      if (keys['KeyA']) moveDir.sub(right);
   562→   562→      if (moveDir.lengthSq() > 0) {
   563→   563→        moveDir.normalize().multiplyScalar(speed);
   564→   564→      }
   565→   565→
   566→   566→      if (sprinting) {
   567→   567→        stateRef.current.stamina = Math.max(0, stateRef.current.stamina - dt * 15);
   568→   568→      }
   569→   569→
   570→   570→      if (player.flying) {
   571→   571→        player.velocity.x = moveDir.x;
   572→   572→        player.velocity.z = moveDir.z;
   573→   573→        player.velocity.y = 0;
   574→   574→        if (keys['Space']) player.velocity.y = WORLD_CONFIG.FLY_SPEED * 0.7;
   575→   575→        if (keys['ShiftLeft']) player.velocity.y = -WORLD_CONFIG.FLY_SPEED * 0.7;
   576→   576→      } else {
   577→   577→        player.velocity.x = moveDir.x;
   578→   578→        player.velocity.z = moveDir.z;
   579→   579→
   580→   580→        // Gravity
   581→   581→        player.velocity.y += WORLD_CONFIG.GRAVITY * dt;
   582→   582→        if (player.velocity.y < -50) player.velocity.y = -50;
   583→   583→
   584→   584→        // Water physics
   585→   585→        if (isInWater(player.position.x, player.position.y, player.position.z)) {
   586→   586→          player.velocity.y = Math.max(player.velocity.y, -3);
   587→   587→          player.velocity.y *= 0.85;
   588→   588→          if (keys['Space']) player.velocity.y = 4;
   589→   589→        }
   590→   590→
   591→   591→        // Jump
   592→   592→        if (keys['Space'] && player.onGround && stateRef.current.stamina > 5) {
   593→   593→          player.velocity.y = WORLD_CONFIG.JUMP_VELOCITY;
   594→   594→          player.onGround = false;
   595→   595→          stateRef.current.stamina -= 3;
   596→   596→        }
   597→   597→      }
   598→   598→
   599→   599→      // Apply movement with collision (per-axis)
   600→   600→      const dx = player.velocity.x * dt;
   601→   601→      const dy = player.velocity.y * dt;
   602→   602→      const dz = player.velocity.z * dt;
   603→   603→
   604→   604→      // X axis
   605→   605→      if (dx !== 0 && !checkCollision(player.position.x + dx, player.position.y, player.position.z)) {
   606→   606→        player.position.x += dx;
   607→   607→      } else {
   608→   608→        player.velocity.x = 0;
   609→   609→      }
   610→   610→      // Z axis
   611→   611→      if (dz !== 0 && !checkCollision(player.position.x, player.position.y, player.position.z + dz)) {
   612→   612→        player.position.z += dz;
   613→   613→      } else {
   614→   614→        player.velocity.z = 0;
   615→   615→      }
   616→   616→      // Y axis
   617→   617→      if (dy !== 0 && !checkCollision(player.position.x, player.position.y + dy, player.position.z)) {
   618→   618→        player.position.y += dy;
   619→   619→        player.onGround = false;
   620→   620→      } else {
   621→   621→        if (dy < 0) player.onGround = true;
   622→   622→        player.velocity.y = 0;
   623→   623→      }
   624→   624→
   625→   625→      // Fall damage / void
   626→   626→      if (player.position.y < -10) {
   627→   627→        stateRef.current.health = 0;
   628→   628→        stateRef.current.dead = true;
   629→   629→        setMessage('Spadłeś w otchłań! Kliknij aby zrespawnować.', 99999);
   630→   630→        syncUI();
   631→   631→      }
   632→   632→
   633→   633→      // Update camera position
   634→   634→      camera.position.set(player.position.x, player.position.y + 1.6, player.position.z);
   635→   635→      const euler = new THREE.Euler(player.pitch, player.yaw, 0, 'YXZ');
