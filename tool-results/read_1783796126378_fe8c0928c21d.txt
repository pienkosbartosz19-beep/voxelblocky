   282→
   283→function useEffectForKeyboard(
   284→  setShowInventory: (v: boolean) => void,
   285→  setShowCrafting: (v: boolean) => void,
   286→  setShowQuests: (v: boolean) => void,
   287→  setShowHelp: (v: boolean) => void,
   288→) {
   289→  useEffect(() => {
   290→    function onKey(e: KeyboardEvent) {
   291→      if (e.code === 'KeyE') {
   292→        e.preventDefault();
   293→        setShowInventory(true);
   294→      } else if (e.code === 'KeyQ') {
   295→        e.preventDefault();
   296→        setShowCrafting(true);
   297→      } else if (e.code === 'Tab') {
   298→        e.preventDefault();
   299→        setShowQuests(true);
   300→      } else if (e.code === 'KeyH') {
   301→        e.preventDefault();
   302→        setShowHelp(true);
   303→      }
   304→    }
   305→    document.addEventListener('keydown', onKey);
   306→    return () => document.removeEventListener('keydown', onKey);
   307→  }, [setShowInventory, setShowCrafting, setShowQuests, setShowHelp]);
   308→}
